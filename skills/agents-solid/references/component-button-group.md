# Button Group

Source: https://shadcn-solid.netlify.app/docs/components/button-group

## Import

```ts
import {
  ButtonGroup,
  ButtonGroupItem,
  ButtonGroupSeparator,
} from "@/components/ui/button-group"
```

## Minimal usage

```tsx
<ButtonGroup>
  <ButtonGroupItem>Left</ButtonGroupItem>
  <ButtonGroupSeparator />
  <ButtonGroupItem>Right</ButtonGroupItem>
</ButtonGroup>
```
